Your API in 5 lines of code

main.go 
package main

import (
    "log"
    "github.com/naufalraism/aemeath"
)

type HelloIn struct {
    Name string `path:"name"`
}

func main() {
    app := aemeath.New()
    
    aemeath.GET(app, "/hello/{name}", func(c *aemeath.Ctx, in HelloIn) (map[string]string, error) {
        return map[string]string{"message": "Hello, " + in.Name}, nil
    })

    log.Fatal(app.Listen(":3000"))
}
